name: Update contribution snake

on:
  # Run once daily at 09:17 China time (01:17 UTC).
  schedule:
    - cron: "17 1 * * *"
  workflow_dispatch:
  # Generate the first images as soon as this file reaches main.
  push:
    branches:
      - main
    paths:
      - .github/workflows/snake.yml

concurrency:
  group: contribution-snake
  cancel-in-progress: false

jobs:
  generate:
    name: Generate and publish contribution images
    runs-on: ubuntu-latest
    timeout-minutes: 10
    permissions:
      contents: write

    steps:
      - name: Generate light and dark SVGs
        uses: Platane/snk/svg-only@v3
        with:
          github_user_name: ${{ github.repository_owner }}
          github_token: ${{ secrets.GITHUB_TOKEN }}
          outputs: |
            dist/github-contribution-grid-snake.svg?color_snake=%230969da
            dist/github-contribution-grid-snake-dark.svg?palette=github-dark&color_snake=%2339d3bb

      # Publish only after generation succeeds; keep previous images on failure.
      - name: Publish SVGs to the output branch
        uses: crazy-max/ghaction-github-pages@df5cc2bfa78282ded844b354faee141f06b41865 # v4.2.0
        with:
          target_branch: output
          build_dir: dist
          commit_message: "chore: update contribution snake"
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
